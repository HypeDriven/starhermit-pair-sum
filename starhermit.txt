# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.24.0
engine.minor=20
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
# The board is also played with the pointer or touch: tap two number tokens
# to connect them; every round action has a button in the tray.
engine.group.left=Board
control.left=ArrowLeft | Move left
control.right=ArrowRight | Move right
control.up=ArrowUp | Move up
control.down=ArrowDown | Move down
control.choose=Enter+Space+NumpadEnter | Select / connect
control.cancel=Escape | Cancel selection / pause / back
engine.group.hint=Round
control.hint=KeyH | Hint
control.undo=KeyU | Undo
control.addrows=KeyA | Add rows
control.camera=KeyR | Reset camera

# Gamepad defaults: the D-pad or left stick moves among the tokens, South
# selects / connects, East and Start cancel or pause, North hints, West
# undoes, the right shoulder adds rows, the right stick press resets the
# camera.
engine.gamepad.left=DpadLeft+LeftStickLeft
engine.gamepad.right=DpadRight+LeftStickRight
engine.gamepad.up=DpadUp+LeftStickUp
engine.gamepad.down=DpadDown+LeftStickDown
engine.gamepad.choose=South
engine.gamepad.cancel=East+Start
engine.gamepad.hint=North
engine.gamepad.undo=West
engine.gamepad.addrows=RightShoulder
engine.gamepad.camera=RightStickPress
